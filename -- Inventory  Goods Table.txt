-- Inventory / Goods Table
CREATE TABLE inventory (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    serial_number VARCHAR(100) UNIQUE NOT NULL,
    product_name VARCHAR(255) NOT NULL,
    category VARCHAR(50) DEFAULT 'Electronics',
    cost_price DECIMAL(10, 2) NOT NULL,
    selling_price DECIMAL(10, 2) NOT NULL,
    stock_quantity INT DEFAULT 1,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Customers Table
CREATE TABLE customers (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    name VARCHAR(255) NOT NULL,
    phone_number VARCHAR(20) UNIQUE NOT NULL,
    address TEXT NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Transactions (Purchases / Sales / Credits)
CREATE TABLE transactions (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    transaction_type VARCHAR(10) CHECK(transaction_type IN ('SALE', 'PURCHASE')),
    payment_type VARCHAR(10) CHECK(payment_type IN ('CASH', 'CREDIT')),
    customer_id INT NULL,
    item_serial_no VARCHAR(100),
    amount DECIMAL(10, 2) NOT NULL,
    transaction_date DATE DEFAULT CURRENT_DATE,
    FOREIGN KEY(customer_id) REFERENCES customers(id),
    FOREIGN KEY(item_serial_no) REFERENCES inventory(serial_number)
);

-- Credit / Ledger Tracking (7-Month Credibility & Interest)
CREATE TABLE credit_ledger (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    transaction_id INT UNIQUE,
    customer_id INT,
    principal_amount DECIMAL(10, 2) NOT NULL,
    current_balance DECIMAL(10, 2) NOT NULL,
    purchase_date DATE DEFAULT CURRENT_DATE,
    due_date DATE NOT NULL, -- Calculated as Purchase Date + 7 Months
    months_overdue INT DEFAULT 0,
    interest_applied DECIMAL(10, 2) DEFAULT 0.00,
    status VARCHAR(20) DEFAULT 'ACTIVE', -- ACTIVE, PAID, OVERDUE
    FOREIGN KEY(transaction_id) REFERENCES transactions(id),
    FOREIGN KEY(customer_id) REFERENCES customers(id)
);
